# Test runner closure (test/dev only, NOT part of the production closure).
#
# The production Dockerfile uninstalls the pip/wheel modules after installing
# the main requirements.txt closure, so a shipped image cannot `pip install`
# ad hoc. The in-image test runner is therefore built in from this *separate*
# closure, installed only when INSTALL_TEST_DEPS_AT_BUILD=true (default false ->
# the production image stays clean and unchanged).
#
# Version pins align to the admin-api test image (pytest 9.1.1 +
# pytest-asyncio 1.4.0 + pygments 2.21.0) so all three services run the same
# runner. The set is the full offline install closure for pytest:
#   pytest          needs pluggy<2,>=1.5, iniconfig>=1.0.1, packaging>=22,
#                  pygments>=2.7.2
#   pytest-asyncio  needs pytest (already here)
# All six are pure-python (`py3-none-any`) wheels, so
# `scripts/pip_wheels_bundle.sh fetch` caches them per arch with
# `pip download --only-binary=:all: --no-deps` (the test closure is a tiny,
# hand-maintained set, unlike the pip-compile `--generate-hashes` production
# closure) and the Dockerfile test layer installs them offline via
# `--no-index --find-links` against that same cache.
#
# pygments is carried by the production closure too (it ships in every service
# image), so listing it here is harmless: offline install sees it in the
# bundle cache and picks the version already installed.

pytest==9.1.1
pytest-asyncio==1.4.0
pluggy==1.6.0
iniconfig==2.3.0
packaging==26.3
pygments==2.21.0
